### 1.2 一条能实际跑的扫描命令（含正则）

```bash
cd /d/Projects/my-pixel-card-game && python -c '
import os,re,collections
RAW=r"\u8be1\u79d8\u4e4b\u4e3b|\u8be1\u79d8|\u6e90\u5821|\u5e8f\u5217|\u611a\u8005|\u9b54\u836f|\u6559\u4f1a|\u5854\u7f57"

def scan(pat,exts=None):
    c=collections.Counter(); f=collections.Counter(); rows=[]
    for dp,_,fs in os.walk("docs"):
        for fn in fs:
            p=os.path.join(dp,fn)
            if exts and not fn.endswith(tuple(exts)): continue
            try: lines=open(p,encoding="utf-8").read().splitlines()
            except Exception: continue
            for i,l in enumerate(lines,1):
                for m in pat.finditer(l):
                    c[m.group()]+=1; f[p]+=1; rows.append((p,i,m.group()))
    return c,f,rows
for name,pat,ex in [("L1_ALL",P1,None),("L1_MD",P1,[".md"]),("L2_ALL",P2,None),("L2_MD",P2,[".md"])]:
    c,f,r=scan(pat,ex)
    print(name,"HITS",sum(c.values()),"FILES",len(f),"BY_WORD",dict(c))
    for p,i,w in r: print("   ",p,i,w)
'

### 3.3 判据：条目级补充优于文件级复制

可机检判定式：

```
duplicate(F) = ( Heads(F) ∩ Heads(当前对应文件) ≠ ∅ )
             ∨ ( F 中条目定义已存在于 schema $defs / effect.oneOf / triggerEvent.enum
